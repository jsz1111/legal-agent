
### 核查

```bash
python -c "
from pymilvus import connections, Collection, utility
from src.core.config import get_settings
s = get_settings()
connections.connect(alias='chk', host=s.MILVUS_HOST, port=s.MILVUS_PORT)
for name in ['statute_index', 'case_index']:
    if utility.has_collection(name, using='chk'):
        c = Collection(name, using='chk'); c.flush()
        print(name, '->', c.num_entities)
connections.disconnect('chk')
"
```
